app.py

import sqlite3
from flask import Flask, jsonify, render_template, request

app = Flask(__name__)
DB = "entries.db"

def init_db():
    with sqlite3.connect(DB) as con:
        con.execute("""
            CREATE TABLE IF NOT EXISTS entries (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                value REAL NOT NULL
            )
        """)
        con.commit()

@app.get("/")
def home():
    return render_template("index.html")

@app.get("/api/entries")
def get_entries():
    with sqlite3.connect(DB) as con:
        con.row_factory = sqlite3.Row
        rows = con.execute("SELECT id, value FROM entries ORDER BY id DESC").fetchall()
    entries = [{"id": r["id"], "value": r["value"]} for r in rows]
    total = sum(e["value"] for e in entries)
    return jsonify({"entries": entries, "total": total})

@app.post("/api/entries")
def add_entry():
    data = request.get_json(force=True)
    try:
        val = float(data["value"])
    except Exception:
        return jsonify({"error": "value must be a number"}), 400

    with sqlite3.connect(DB) as con:
        con.execute("INSERT INTO entries(value) VALUES (?)", (val,))
        con.commit()
    return jsonify({"ok": True})

if __name__ == "__main__":
    init_db()
    app.run(host="0.0.0.0", port=8000, debug=True)

<!doctype html>
<html>
<head>
  <meta charset="utf-8" />
  <meta name="viewport" content="width=device-width,initial-scale=1" />
  <title>Float Logger</title>
  <style>
    body { font-family: system-ui; margin: 16px; max-width: 720px; }
    input, button { font-size: 18px; padding: 10px; }
    .row { display:flex; gap:10px; align-items:center; flex-wrap:wrap; }
    #total { font-weight: 800; margin-top: 14px; }
    ul { padding-left: 18px; }
  </style>
</head>
<body>
  <h2>Float Logger</h2>

  <div class="row">
    <input id="val" inputmode="decimal" placeholder="Enter a number" />
    <button id="add">Add</button>
  </div>

  <div id="total">Total: 0.00</div>

  <h3>Entries</h3>
  <ul id="list"></ul>

  <script>
    const valEl = document.getElementById("val");
    const totalEl = document.getElementById("total");
    const listEl = document.getElementById("list");

    function fmt2(x) { return Number(x).toFixed(2); }

    async function refresh() {
      const res = await fetch("/api/entries");
      const data = await res.json();
      totalEl.textContent = "Total: " + fmt2(data.total);

      listEl.innerHTML = "";
      for (const e of data.entries) {
        const li = document.createElement("li");
        li.textContent = e.value;
        listEl.appendChild(li);
      }
    }

    document.getElementById("add").addEventListener("click", async () => {
      const raw = valEl.value.trim();
      if (!raw) return;

      const n = Number(raw);
      if (!Number.isFinite(n)) {
        alert("Please enter a valid number");
        return;
      }

      await fetch("/api/entries", {
        method: "POST",
        headers: {"Content-Type":"application/json"},
        body: JSON.stringify({ value: n })
      });

      valEl.value = "";
      await refresh();
    });

    // load on page open
    refresh();
  </script>
</body>
</html>
